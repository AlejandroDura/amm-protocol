    function addLiquidity2(uint256 _t0Deposit, uint256 _t1Deposit) external {
        if (_t0Deposit == 0 || _t1Deposit == 0) {
            revert Pool__NewReservesMustBeGreaterThanZero(_t0Deposit, _t1Deposit);
        }

        uint256 token0PrevBalance = ERC20(s_token0).balanceOf(address(this));
        uint256 token1PrevBalance = ERC20(s_token1).balanceOf(address(this));

        bool success = IERC20(s_token0).transferFrom(msg.sender, address(this), _t0Deposit);
        if (!success) {
            revert Pool__TransferFailed(s_token0);
        }

        success = IERC20(s_token1).transferFrom(msg.sender, address(this), _t1Deposit);
        if (!success) {
            revert Pool__TransferFailed(s_token1);
        }

        uint256 token0CurrentBalance = ERC20(s_token0).balanceOf(address(this));
        uint256 token1CurrentBalance = ERC20(s_token1).balanceOf(address(this));

        _t0Deposit = token0CurrentBalance - token0PrevBalance;
        _t1Deposit = token1CurrentBalance - token1PrevBalance;

        (uint256 t0AmountUsed, uint256 t1AmountUsed) = _mintLP(_t0Deposit, _t1Deposit);

        s_reserveToken0 += t0AmountUsed;
        s_reserveToken1 += t1AmountUsed;

        success = ERC20(s_token0).transfer(msg.sender, _t0Deposit - t0AmountUsed);
        if (!success) {
            revert Pool__TransferFailed(s_token0);
        }

        success = ERC20(s_token1).transfer(msg.sender, _t1Deposit - t1AmountUsed);
        if (!success) {
            revert Pool__TransferFailed(s_token1);
        }
    }

        function removeLiquidity(uint256 _userIndex, uint256 _liquidityToRetrieve) public {
        uint256 totalLiquidity = pool.getTotalLP();
        if (totalLiquidity == 0) return;

        address user = users[_userIndex % users.length];
        uint256 userLP = pool.getUserLP(user);
        if (userLP == 0) return;

        uint256 liquiditySanitized = bound(_liquidityToRetrieve, 1, userLP);

        (uint256 reserve0, uint256 reserve1) = pool.getReservePairs();

        uint256 expectedT0toRetrieve = (liquiditySanitized * reserve0) / totalLiquidity;
        uint256 expectedT1toRetrieve = (liquiditySanitized * reserve1) / totalLiquidity;

        uint256 userToken0PrevBalance = ERC20Mock(token0).balanceOf(user);
        uint256 userToken1PrevBalance = ERC20Mock(token1).balanceOf(user);

        vm.startPrank(user);
        pool.removeLiquidity(liquiditySanitized);
        vm.stopPrank();

        uint256 userToken0CurrentBalance = ERC20Mock(token0).balanceOf(user);
        uint256 userToken1CurrentBalance = ERC20Mock(token1).balanceOf(user);

        assertEq(userToken0CurrentBalance - userToken0PrevBalance, expectedT0toRetrieve, "T0 TO RETRIEVE FAIL!!");
        assertEq(userToken1CurrentBalance - userToken1PrevBalance, expectedT1toRetrieve, "T1 TO RETRIEVE FAIL!!");
        assertEq(pool.getUserLP(user), userLP - liquiditySanitized);
        assertEq(pool.getTotalLP(), totalLiquidity - liquiditySanitized);
    }
